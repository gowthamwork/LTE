import serial
import keyboard
import time

# Replace with your Pico's COM port (e.g., COM3 or /dev/ttyACM0)
pico = serial.Serial('COM4', 9600)

print("Press UP to run motor, DOWN to stop. Press Q to quit.")

while True:
    if keyboard.is_pressed("up"):
        pico.write(b'F')
        print("Sent: F")
        time.sleep(0.2)

    elif keyboard.is_pressed("down"):
        pico.write(b'S')
        print("Sent: S")
        time.sleep(0.2)

    elif keyboard.is_pressed("q"):
        pico.write(b'S')
        print("Quitting...")
        break
