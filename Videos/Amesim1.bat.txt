@echo off 
echo This is an example on how to start a Simcenter Amesim version which
echo is different from the latest one. You can use this,
echo for instance, to start Simcenter Amesim Rev M on a machine where you have
echo installed Simcenter Amesim Rev N. Lines after "CHANGE THIS" are likely 
echo to be changed.
 
REM The setlocal makes any changes to environment variables local 
REM between here and the endlocal flag.
 
setlocal
 
REM Here are defined environment variables for MSVC++.
 
call "%AME%\vcvars32.bat"
 
REM We define the AME environment variable:
REM this will of course be different for each usage.
 
REM ***********
REM CHANGE THIS
REM ***********
set AME=C:\Program Files\Simcenter\2404\Amesim
set PATH=%AME%\sys\mingw64\bin;%PATH%
 
REM We can here also change directory to where we want to start
REM for instance "D:\AMETest".
 
REM ***********
REM CHANGE THIS
REM ***********
cd D:\AMETest
 
REM ********************************************
REM THIS SECTION IS IMPORTANT FOR EXCEL PLUGIN
REM ********************************************
set QT_PLUGIN_PATH=%AME%\win64
set QTWEBENGINEPROCESS_PATH=%AME%\win64
REM ACTIVATE BELOW LINE IF YOU ARE USING EXCEL 32 bit version
REM regsvr32 %AME%\win32\ame_treeview_control32.dll
REM Uninstall Excel plugin
REM regsvr32 /u %AME%\win32\ame_treeview_control32.dll
 
REM ACTIVATE if using Microsoft Office 64-Bit version
REM Install
REM regsvr32 %AME%\win64\ame_treeview_control64.dll
REM Uninstall
REM regsvr32 /u %AME%\win64\ame_treeview_control64.dll
 
 
REM We also define licensing environment variables,
REM based on RLM (default) or FlexNET (SPLM for FlexNET since Simcenter Amesim 2020.1, AMED for FlexNet with older Simcenter Amesim versions).
REM ***********
REM CHANGE THIS
REM ***********
REM set AMED_LICENSE_FILE=port@server
set LMS_LICENSE=5053@74.225.193.188
REM set SPLM_LICENSE_SERVER=29000@LocalHost
REM For RLM set 'rlm' (default), for FlexNET set 'ugs'. Any other value will use RLM by default.
SET SC1D_LICENSING_TYPE=rlm
 
REM then we set up the path to include the Simcenter Amesim related directories.
if exist "%AME%\Common" set AME=%AME%\Amesim
if exist "%AME%\Common" set SYSSYN=%AME%\System Architect
 
set AME_CMMN=%AME%\Common
set PYTHONPATH=%PYTHONPATH%;%AME%\scripting\python;%AME%\scripting\python\win64;C:\PythonPackages
 
if exist "%SYSSYN%\win64" set path=%SYSSYN%\win64;%PATH%
if exist "%SYSSYN%\win32" set path=%SYSSYN%\win32;%PATH%
if exist "%AME_CMMN%\win32" set path=%AME_CMMN%\win32;%path%
set path=%AME%;%AME%\win32;%AME%\win64;%AME%\sys\mingw32\bin;%AME%\sys\mingw64\bin;%AME%\sys\python\win32;%AME%\sys\python\win64;%path%
 
REM %AME%\sys\python\win64
REM Finally start Simcenter Amesim, using the AME environment variable to make
REM sure that we start the Simcenter Amesim version we want to start.
if exist "%AME%\win64\AMESim.exe" start "Amesim" "%AME%\win64\AMESim.exe"
if exist "%AME%\win32\AMESim.exe" start "Amesim" "%AME%\win32\AMESim.exe"
 
 
endlocal
 
echo THIS WINDOW WILL CLOSE IN 10 SECOND
TIMEOUT /T 10
exit
REM Make the DOS window wait for a key press, to catch any error printouts.
